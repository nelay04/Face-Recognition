# ---- Vision / numerics ----
# dlib-bin ships prebuilt wheels (incl. cp314). Never depend on `dlib`
# itself: it is source-only and forces a slow, fragile CMake build.
dlib-bin==20.0.1
numpy==2.5.2
opencv-python-headless==5.0.0.93
pillow==12.3.0

# ---- Pretrained model weights ----
face-recognition-models @ git+https://github.com/ageitgey/face_recognition_models

# `face_recognition_models` imports `pkg_resources` at module load. That module
# is no longer bundled with Python 3.12+ venvs, and setuptools removes it in 81.
# Without this pin, `import face_recognition` fails with a *misleading* message
# telling you to reinstall the models package.
setuptools<81
